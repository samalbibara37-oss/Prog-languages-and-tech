<script>
  // Ждем полной загрузки HTML, прежде чем запускать скрипт
  document.addEventListener('DOMContentLoaded', () => {
    
    const STORAGE_KEY = 'personal_questionnaire_data_v2'; // Изменили ключ, чтобы сбросить старые сломанные данные

    const form = document.getElementById('profileForm');
    const nameInput = document.getElementById('name');
    const groupInput = document.getElementById('group');
    const cityInput = document.getElementById('city');
    const resetBtn = document.getElementById('resetBtn');
    const saveStatus = document.getElementById('saveStatus');

    let statusTimeout;

    // 1. Восстановление данных
    function loadData() {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved) {
        try {
          const data = JSON.parse(saved);
          nameInput.value = data.name || '';
          groupInput.value = data.group || '';
          cityInput.value = data.city || '';
        } catch (e) {
          console.error('Ошибка чтения данных', e);
        }
      }
    }

    // 2. Сохранение данных
    function saveData() {
      const data = {
        name: nameInput.value.trim(),
        group: groupInput.value.trim(),
        city: cityInput.value.trim()
      };
      
      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
      
      // Показываем сообщение
      saveStatus.classList.add('visible');
      clearTimeout(statusTimeout);
      statusTimeout = setTimeout(() => {
        saveStatus.classList.remove('visible');
      }, 2500);
    }

    // 3. Очистка
    function resetData() {
      nameInput.value = '';
      groupInput.value = '';
      cityInput.value = '';
      localStorage.removeItem(STORAGE_KEY);
      saveStatus.classList.remove('visible');
    }

    // --- Слушатели событий ---

    // Сохранение при нажатии на кнопку "Сохранить" или Enter
    form.addEventListener('submit', (e) => {
      e.preventDefault(); // Останавливаем перезагрузку страницы
      saveData();
    });

    // Очистка
    resetBtn.addEventListener('click', resetData);

    // Автосохранение при вводе (чтобы данные не терялись, даже если не нажать кнопку)
    nameInput.addEventListener('input', saveData);
    groupInput.addEventListener('input', saveData);
    cityInput.addEventListener('input', saveData);

    // Запускаем восстановление при открытии
    loadData();
  });
</script>
