<script>
  // ==========================================
  // ЯДРО ПРИЛОЖЕНИЯ (Обернуто в DOMContentLoaded)
  // ==========================================
  document.addEventListener('DOMContentLoaded', () => {
    
    // Ключ для хранения. Версия v3, чтобы сбросить старые баги.
    const STORAGE_KEY = 'room_booking_app_v3';
    
    // Элементы DOM
    const form = document.getElementById('bookingForm');
    const roomSelect = document.getElementById('room');
    const dateInput = document.getElementById('date');
    const startTimeInput = document.getElementById('startTime');
    const endTimeInput = document.getElementById('endTime');
    const bookingList = document.getElementById('bookingList');
    const scheduleDate = document.getElementById('scheduleDate');
    const alertBox = document.getElementById('alertBox');

    // Глобальный массив записей
    let bookings = [];

    // --- 1. ИНИЦИАЛИЗАЦИЯ И ЗАГРУЗКА ---
    function init() {
      const todayStr = new Date().toISOString().split('T')[0];
      dateInput.value = todayStr;
      dateInput.min = todayStr;

      // Загружаем из памяти
      const saved = localStorage.getItem(STORAGE_KEY);
      
      if (saved) {
        try {
          bookings = JSON.parse(saved);
          if (bookings.length > 0) {
            showAlert(`Успешно загружено ${bookings.length} записей из памяти.`, 'success');
          }
        } catch (e) {
          console.error("Ошибка парсинга", e);
          bookings = [];
        }
      } else {
        // СОЗДАЕМ ТЕСТОВУЮ ЗАПИСЬ, ЧТОБЫ ДОКАЗАТЬ, ЧТО СОХРАНЕНИЕ РАБОТАЕТ
        bookings = [{
          id: "demo-1",
          room: "101",
          date: todayStr,
          start: "10:00",
          end: "12:00"
        }];
        saveToStorage();
      }

      renderSchedule();
    }

    // --- 2. СОХРАНЕНИЕ ---
    function saveToStorage() {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(bookings));
      } catch (e) {
        showAlert("Ошибка! Браузер запретил сохранение данных.", "error");
      }
    }

    // --- 3. УВЕДОМЛЕНИЯ ---
    function showAlert(message, type) {
      alertBox.textContent = message;
      alertBox.className = `alert ${type}`;
      
      // Очистка таймера при повторном вызове
      if (window.alertTimeout) clearTimeout(window.alertTimeout);
      
      // Скрываем через 5 секунд, если это успех
      if (type === 'success') {
        window.alertTimeout = setTimeout(() => {
          alertBox.className = 'alert';
        }, 5000);
      }
    }

    // --- 4. ЛОГИКА КОНФЛИКТОВ ---
    // Проверяет, пересекается ли новое время с уже существующими записями
    function findConflict(room, date, newStart, newEnd) {
      // Ищем записи только для этой комнаты и этой даты
      const sameDayBookings = bookings.filter(b => b.room === room && b.date === date);

      for (let i = 0; i < sameDayBookings.length; i++) {
        const existing = sameDayBookings[i];
        
        // МАГИЯ ПЕРЕСЕЧЕНИЯ ИНТЕРВАЛОВ:
        // Конфликт есть, если (Новое начало < Существующего конца) И (Новый конец > Существующего начала)
        // Строки формата "HH:MM" отлично сравниваются лексикографически в JS.
        if (newStart < existing.end && newEnd > existing.start) {
          return existing; // Возвращаем запись, с которой конфликт
        }
      }
      return null; // Конфликтов нет
    }

    // --- 5. ОТРИСОВКА СПИСКА ---
    function renderSchedule() {
      const currentRoom = roomSelect.value;
      const currentDate = dateInput.value;

      // Форматируем дату для красоты
      const [y, m, d] = currentDate.split('-');
      scheduleDate.textContent = currentDate === new Date().toISOString().split('T')[0] 
        ? 'Сегодня' : `${d}.${m}.${y}`;

      // Фильтруем по комнате и дате
      const filtered = bookings.filter(b => b.room === currentRoom && b.date === currentDate);
      
      // Сортируем по времени начала
      filtered.sort((a, b) => a.start.localeCompare(b.start));

      bookingList.innerHTML = '';

      if (filtered.length === 0) {
        bookingList.innerHTML = `<div class="empty-state">На это время комната полностью свободна.</div>`;
        return;
      }

      filtered.forEach(booking => {
        const card = document.createElement('div');
        card.className = 'booking-card';
        card.innerHTML = `
          <div class="booking-info">
            <h4>Аудитория ${booking.room}</h4>
            <div class="booking-time">${booking.start} — ${booking.end}</div>
          </div>
          <button class="btn-delete" data-id="${booking.id}" title="Удалить бронь">✕</button>
        `;
        bookingList.appendChild(card);
      });
    }

    // --- 6. ОБРАБОТКА ФОРМЫ (ДОБАВЛЕНИЕ) ---
    form.addEventListener('submit', (e) => {
      e.preventDefault();

      const room = roomSelect.value;
      const date = dateInput.value;
      const start = startTimeInput.value;
      const end = endTimeInput.value;

      // Базовая валидация
      if (start >= end) {
        showAlert('Ошибка: Время окончания должно быть строго позже времени начала.', 'error');
        return;
      }

      // Проверка на конфликт
      const conflict = findConflict(room, date, start, end);

      if (conflict) {
        showAlert(`КОНФЛИКТ! Аудитория ${room} уже занята с ${conflict.start} до ${conflict.end}. Выберите другое время.`, 'error');
        return;
      }

      // Если всё ок — создаем запись
      const newBooking = {
        id: 'booking_' + Date.now(),
        room: room,
        date: date,
        start: start,
        end: end
      };

      bookings.push(newBooking);
      saveToStorage();
      renderSchedule();

      showAlert('Успешно! Комната забронирована и запись сохранена.', 'success');
      
      // Очищаем время
      startTimeInput.value = '';
      endTimeInput.value = '';
    });

    // --- 7. УДАЛЕНИЕ ---
    bookingList.addEventListener('click', (e) => {
      if (e.target.classList.contains('btn-delete')) {
        const id = e.target.getAttribute('data-id');
        
        if (confirm('Вы уверены, что хотите отменить эту бронь?')) {
          bookings = bookings.filter(b => b.id !== id);
          saveToStorage();
          renderSchedule();
          showAlert('Бронь отменена.', 'success');
        }
      }
    });

    // --- 8. СМЕНА ФИЛЬТРОВ ---
    // Если пользователь меняет комнату или дату в форме, список справа обновляется
    roomSelect.addEventListener('change', renderSchedule);
    dateInput.addEventListener('change', renderSchedule);

    // Запуск приложения
    init();
  });
</script>
