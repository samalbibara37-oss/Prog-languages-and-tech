const content = document.getElementById("content");
const tabs = document.querySelectorAll(".tab");
const searchInput = document.getElementById("searchInput");
const searchButton = document.getElementById("searchButton");
const resetButton = document.getElementById("resetButton");

let currentTab = "news";

function renderMaterials() {
    const keyword = searchInput.value.trim().toLowerCase();

    const filteredData = facultyData[currentTab].filter(item => {
        const text = (
            item.title + " " +
            item.date + " " +
            item.text
        ).toLowerCase();

        return text.includes(keyword);
    });

    if (filteredData.length === 0) {
        content.innerHTML =
            '<p class="empty">Материалы не найдены.</p>';
        return;
    }

    content.innerHTML = filteredData.map(item => `
        <article class="card">
            <h3>${item.title}</h3>
            <small>Дата: ${item.date}</small>
            <p>${item.text}</p>
        </article>
    `).join("");
}

function switchTab(tabName) {
    currentTab = tabName;

    tabs.forEach(tab => {
        tab.classList.toggle(
            "active",
            tab.dataset.tab === tabName
        );
    });

    renderMaterials();
}

tabs.forEach(tab => {
    tab.addEventListener("click", () => {
        switchTab(tab.dataset.tab);
    });
});

searchButton.addEventListener("click", renderMaterials);

searchInput.addEventListener("input", renderMaterials);

resetButton.addEventListener("click", () => {
    searchInput.value = "";
    renderMaterials();
});

switchTab("news");
